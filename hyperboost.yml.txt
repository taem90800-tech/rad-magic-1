          /** แถบปัดข้างสไตล์ HyperBoost: ปัดจากขอบซ้ายเพื่อเปิดเมนูขณะเล่นเกม */
          class OverlayService : Service() {
              private lateinit var wm: WindowManager
              private lateinit var prefs: SharedPreferences
              private val h = Handler(Looper.getMainLooper())
              private var handle: View? = null
              private var root: FrameLayout? = null
              private var panel: View? = null
              private var navBox: LinearLayout? = null
              private var content: FrameLayout? = null
              private var tab = 0
              private var fpsGauge: ArcGauge? = null
              private var pingGauge: ArcGauge? = null
              private var batTv: TextView? = null

              @Volatile private var alive = false

              companion object { @Volatile var running = false }

              private val OR = Color.parseColor("#FF5A14")
              private val GRAY = 0xFF8A8A8A.toInt()
              private val WHITE = Color.WHITE
              private val vKeys = listOf("normal", "kid", "deep", "robot", "echo")
              private val vNames = listOf("ปกติ", "เสียงเด็ก", "เสียงทุ้ม", "หุ่นยนต์", "เสียงก้อง")
              private val shortcuts = listOf("com.instagram.android", "com.zhiliaoapp.musically", "com.google.android.youtube")

              private val tick = object : Runnable {
                  override fun run() { updateStats(); h.postDelayed(this, 1500) }
              }

              private fun dp(v: Int) = (v * resources.displayMetrics.density).toInt()
              private fun round(c: Int, r: Int) = GradientDrawable().apply { setColor(c); cornerRadius = r.toFloat() }
              private fun toast(t: String) = Toast.makeText(this, t, Toast.LENGTH_SHORT).show()
              private fun lab(t: String, sp: Float, c: Int = WHITE) = TextView(this).apply { text = t; textSize = sp; setTextColor(c) }
              private fun divider() = View(this).apply {
                  setBackgroundColor(Color.parseColor("#333333"))
                  layoutParams = LinearLayout.LayoutParams(-1, dp(1)).apply { topMargin = dp(12); bottomMargin = dp(12) }
              }

              override fun onStartCommand(i: Intent?, flags: Int, id: Int): Int {
                  if (i?.action == "stop") { stopSelf(); return START_NOT_STICKY }
                  val nm = getSystemService(NOTIFICATION_SERVICE) as NotificationManager
                  nm.createNotificationChannel(NotificationChannel("ga", "ผู้ช่วยเกม", NotificationManager.IMPORTANCE_LOW))
                  val n = Notification.Builder(this, "ga").setSmallIcon(R.drawable.ic_launcher)
                      .setContentTitle("ผู้ช่วยเกม").setContentText("ปัดจากขอบซ้ายเพื่อเปิดเมนู").build()
                  if (Build.VERSION.SDK_INT >= 34)
                      startForeground(2, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_SPECIAL_USE)
                  else startForeground(2, n)
                  if (handle == null) {
                      wm = getSystemService(WINDOW_SERVICE) as WindowManager
                      prefs = getSharedPreferences("ga", MODE_PRIVATE)
                      alive = true
                      addHandle(); startPing(); h.post(tick)
                  }
                  running = true
                  return START_NOT_STICKY
              }

              // ---------- ขอบปัด ----------
              private fun addHandle() {
                  val r = dp(8).toFloat()
                  val v = View(this).apply {
                      background = GradientDrawable().apply {
                          setColor(Color.parseColor("#80FF5A14"))
                          cornerRadii = floatArrayOf(0f, 0f, r, r, r, r, 0f, 0f)
                      }
                  }
                  var sx = 0f
                  v.setOnTouchListener { _, e ->
                      when (e.action) {
                          MotionEvent.ACTION_DOWN -> sx = e.rawX
                          MotionEvent.ACTION_MOVE -> if (sx != Float.MAX_VALUE && e.rawX - sx > dp(24)) { sx = Float.MAX_VALUE; openPanel() }
                          MotionEvent.ACTION_UP -> { if (sx != Float.MAX_VALUE && abs(e.rawX - sx) < dp(8)) openPanel() }
                      }
                      true
                  }
                  val lp = WindowManager.LayoutParams(dp(22), dp(120), WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
                      WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE, PixelFormat.TRANSLUCENT
                  ).apply { gravity = Gravity.START or Gravity.CENTER_VERTICAL }
                  wm.addView(v, lp); handle = v
              }

              // ---------- แผงเมนู ----------
              private fun openPanel() {
                  if (root != null) return
                  val pw = minOf(dp(340), (resources.displayMetrics.widthPixels * 0.92).toInt())
                  val scrim = FrameLayout(this).apply {
                      setBackgroundColor(Color.parseColor("#66000000")); setOnClickListener { closePanel() }
                  }
                  val p = LinearLayout(this).apply {
                      orientation = LinearLayout.HORIZONTAL; background = round(Color.parseColor("#1C1C1C"), dp(28))
                      isClickable = true
                  }
                  navBox = LinearLayout(this).apply {
                      orientation = LinearLayout.VERTICAL; gravity = Gravity.CENTER_HORIZONTAL
                      background = round(Color.parseColor("#242424"), dp(28))
                  }
                  content = FrameLayout(this)
                  p.addView(navBox, LinearLayout.LayoutParams(dp(76), -1))
                  p.addView(content, LinearLayout.LayoutParams(0, -1, 1f))
                  scrim.addView(p, FrameLayout.LayoutParams(pw, -1).apply { gravity = Gravity.START; setMargins(dp(8), dp(8), 0, dp(8)) })
                  buildNav(); showPage()
                  val lp = WindowManager.LayoutParams(-1, -1, WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
                      WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN,
                      PixelFormat.TRANSLUCENT)
                  wm.addView(scrim, lp); root = scrim; panel = p
                  p.translationX = -(pw + dp(16)).toFloat()
                  p.animate().translationX(0f).setDuration(220).start()
              }

              private fun closePanel() {
                  val r = root ?: return
                  val p = panel
                  root = null; panel = null
                  val remove = Runnable { try { wm.removeView(r) } catch (e: Exception) {} }
                  if (p != null) p.animate().translationX(-(p.width + dp(16)).toFloat()).setDuration(180).withEndAction(remove).start()
                  else remove.run()
              }

              private fun buildNav() {

                  items.forEachIndexed { i, (ic, name) ->
                      val c = if (i == tab) OR else GRAY
                      val item = LinearLayout(this).apply {
                          orientation = LinearLayout.VERTICAL; gravity = Gravity.CENTER_HORIZONTAL
                          setPadding(0, dp(14), 0, dp(6)); setOnClickListener { tab = i; buildNav(); showPage() }
                      }
                      item.addView(ImageView(this).apply { setImageResource(ic); setColorFilter(c, PorterDuff.Mode.SRC_IN) },
                          LinearLayout.LayoutParams(dp(30), dp(30)))
                      item.addView(lab(name, 11f, c).apply { gravity = Gravity.CENTER; setTypeface(null, Typeface.BOLD) })
                      nb.addView(item, LinearLayout.LayoutParams(-1, -2))
                  }
                  nb.addView(View(this).apply { setBackgroundColor(Color.parseColor("#3A3A3A")) },
                      LinearLayout.LayoutParams(dp(40), dp(1)).apply { topMargin = dp(12); bottomMargin = dp(8) })
                  val pm = packageManager
                  shortcuts.filter { pk -> try { pm.getPackageInfo(pk, 0); true } catch (e: Exception) { false } }.forEach { pk ->
                      val iv = ImageView(this).apply {
                          setImageDrawable(pm.getApplicationIcon(pk)); scaleType = ImageView.ScaleType.CENTER_CROP
                          background = GradientDrawable().apply { shape = GradientDrawable.OVAL; setColor(Color.DKGRAY) }
                          outlineProvider = ViewOutlineProvider.BACKGROUND; clipToOutline = true
                          setOnClickListener {
                              pm.getLaunchIntentForPackage(pk)?.let { startActivity(it.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)) }

                      nb.addView(iv, LinearLayout.LayoutParams(dp(46), dp(46)).apply { topMargin = dp(8) })
                  }

              private fun showPage() {

              private fun segmented(opts: List<String>, sel: Int, onSel: (Int) -> Unit): LinearLayout {
                  val box = LinearLayout(this).apply {
                      setPadding(dp(4), dp(4), dp(4), dp(4)); background = round(Color.parseColor("#2A2A2A"), dp(40))
                  }
                  fun paint(i: Int) {
                      for (k in 0 until box.childCount) {
                          val tv = box.getChildAt(k) as TextView
                          tv.background = if (k == i) round(OR, dp(40)) else null
                          tv.setTextColor(if (k == i) WHITE else 0xFFBBBBBB.toInt())
                      }
                  }
                  opts.forEachIndexed { i, s ->
                      box.addView(TextView(this).apply {
                          text = s; gravity = Gravity.CENTER; textSize = 12f; maxLines = 1; ellipsize = TextUtils.TruncateAt.END
                          setPadding(dp(4), dp(10), dp(4), dp(10)); setTypeface(null, Typeface.BOLD)
                          setOnClickListener { paint(i); onSel(i) }
                      }, LinearLayout.LayoutParams(0, -2, 1f))
                  }
                  paint(sel)
                  return box
              }

          package com.example.gameassistant

          import android.annotation.SuppressLint
          import android.media.*
          import kotlin.math.PI
          import kotlin.math.sin

          /** รับเสียงจากไมค์ แล้วเล่นออกทันทีพร้อมเอฟเฟกต์ (ควรใช้หูฟังเพื่อกันเสียงหอน) */
          class VoiceEngine {
              @Volatile var mode = "normal"
              @Volatile private var running = false
              private var thread: Thread? = null

              @SuppressLint("MissingPermission")
              fun start() {
                  if (running) return
                  running = true
                  thread = Thread {
                      val sr = 16000
                      val minIn = AudioRecord.getMinBufferSize(sr, AudioFormat.CHANNEL_IN_MONO, AudioFormat.ENCODING_PCM_16BIT)
                      val rec = AudioRecord(MediaRecorder.AudioSource.VOICE_COMMUNICATION, sr,
                          AudioFormat.CHANNEL_IN_MONO, AudioFormat.ENCODING_PCM_16BIT, minIn * 2)
                      val minOut = AudioTrack.getMinBufferSize(sr, AudioFormat.CHANNEL_OUT_MONO, AudioFormat.ENCODING_PCM_16BIT)
                      val track = AudioTrack.Builder()
                          .setAudioAttributes(AudioAttributes.Builder()
                              .setUsage(AudioAttributes.USAGE_VOICE_COMMUNICATION)
                              .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH).build())
                          .setAudioFormat(AudioFormat.Builder().setSampleRate(sr)
                              .setEncoding(AudioFormat.ENCODING_PCM_16BIT)
                              .setChannelMask(AudioFormat.CHANNEL_OUT_MONO).build())
                          .setBufferSizeInBytes(minOut * 2)
                          .setTransferMode(AudioTrack.MODE_STREAM).build()
                      val buf = ShortArray(512)
                      val echo = ShortArray(sr / 4)
                      var ei = 0
                      var t = 0L
                      rec.startRecording(); track.play()
                      while (running) {
                          val n = rec.read(buf, 0, buf.size)
                          if (n <= 0) continue
                          val m = mode
                          val f = when (m) { "kid" -> 1.4; "deep" -> 0.75; else -> 1.0 }
                          val len = (n / f).toInt().coerceAtLeast(1)
                          val out = ShortArray(len)
                          for (i in 0 until len) {
                              val p = i * f
                              val a = p.toInt().coerceAtMost(n - 1)
                              val b = (a + 1).coerceAtMost(n - 1)
                              var s = buf[a] + (buf[b] - buf[a]) * (p - a)
                              if (m == "robot") s *= sin(2 * PI * 50 * (t++) / sr)
                              if (m == "echo") {
                                  s += echo[ei] * 0.45
                                  echo[ei] = s.coerceIn(-32768.0, 32767.0).toInt().toShort()
                                  ei = (ei + 1) % echo.size
                              }
                              out[i] = s.coerceIn(-32768.0, 32767.0).toInt().toShort()
                          }
                          track.write(out, 0, len)
                      }
                      rec.stop(); rec.release(); track.stop(); track.release()
                  }.also { it.start() }
              }

              fun stop() { running = false; thread?.join(500); thread = null }
          }
